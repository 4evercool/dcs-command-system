## Скрипт среза контракта — фиксирован дословно, «до» и «после» одним и тем же

```python
import pathlib, re, sys
s = pathlib.Path('dcs/references/schemas.md').read_bytes().replace(b'\r\n', b'\n').decode()
sec, out, fenced = None, [], False
for line in s.split('\n'):
    if line.startswith('```'):
        fenced = not fenced; continue
    if fenced:
        continue
    m = re.match(r'^## (\d+)\.', line)
    if m:
        sec = m.group(1)
    if line.startswith('|') and sec and not re.match(r'^\|[\s:-]+\|', line):
        cells = [c.strip() for c in line.split('|')[1:-1]]
        if len(cells) >= 2:
            out.append(f'#{sec}\t{cells[0]}\t{cells[1]}')
sys.stdout.write('\n'.join(out) + '\n')
sys.stderr.write(f'{len(out)} rows\n')
```
